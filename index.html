Google Apps Script Sync Setup

Connect a free Google Apps Script web app attached to a Google Sheet to share live attendance records across all users in real time.

Google Apps Script Code (Code.gs) 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var attSheet = getOrCreateSheet(ss, "Attendance");
  var userSheet = getOrCreateSheet(ss, "Users");
  
  var attendanceData = attSheet.getDataRange().getValues();
  var userData = userSheet.getDataRange().getValues();
  
  var attendance = [];
  for (var i = 1; i < attendanceData.length; i++) {
    if (attendanceData[i][0]) {
      attendance.push({
        user: String(attendanceData[i][0]).toLowerCase().trim(),
        date: attendanceData[i][1],
        status: attendanceData[i][2]
      });
    }
  }
  
  var users = [];
  for (var j = 1; j < userData.length; j++) {
    if (userData[j][0]) {
      users.push({
        user: String(userData[j][0]).toLowerCase().trim(),
        color: userData[j][1] || 'emerald'
      });
    }
  }
  
  var result = { attendance: attendance, users: users };
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var attSheet = getOrCreateSheet(ss, "Attendance");
    var userSheet = getOrCreateSheet(ss, "Users");
    
    var username = String(data.user).toLowerCase().trim();
    var date = data.date;
    var status = data.status || "PRESENT";
    var color = data.color || "emerald";
    
    updateUserColor(userSheet, username, color);
    updateAttendance(attSheet, username, date, status);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (name === "Attendance") {
      sheet.appendRow(["User", "Date", "Status"]);
    } else if (name === "Users") {
      sheet.appendRow(["User", "Color"]);
    }
  }
  return sheet;
}

function updateUserColor(sheet, user, color) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase().trim() === user) {
      sheet.getRange(i + 1, 2).setValue(color);
      return;
    }
  }
  sheet.appendRow([user, color]);
}

function updateAttendance(sheet, user, date, status) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase().trim() === user && data[i][1] === date) {
      if (status === "ABSENT") {
        sheet.deleteRow(i + 1);
      } else {
        sheet.getRange(i + 1, 3).setValue(status);
      }
      return;
    }
  }
  if (status === "PRESENT") {
    sheet.appendRow([user, date, "PRESENT"]);
  }
}
